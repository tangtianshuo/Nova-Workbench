```typescript
interface ProductMilestone {
  id?: string;
  title: string;
  date: string;
  stage?: string;
  status: 'pending' | 'in-progress' | 'completed';
  owner?: string;
  deliverables?: string[];      // 旧:自由字符串,如 ['PRD v3.0']
  deliverableCodes?: string[];  // 🆕 L5: 引用 FullLifecycleDeliverable.code,如 ['DEL-REQ-01']
  description?: string;
}

// rndStore (07-01 Task 2 扩展):
getDeliverablesForProduct(productId: string): FullLifecycleDeliverable[];
getDeliverableStatusForPhase(
  productId: string,
  phase: 'requirement' | 'design' | 'dev' | 'test' | 'release'
): { total: number; ready: number; generating: number; draft: number };

// FullLifecycleDeliverable.code 映射 (mockRndData.ts CATALOG):
// DEL-REQ-01..04 → requirement phase (4 项)
// DEL-DES-01..02 → design phase (2 项)
// DEL-DEV-01..03 → dev phase (3 项)
// DEL-TST-01..03 → test phase (3 项)
// DEL-REL-01..04 → release phase (4 项)

// Product.stage (mockProducts.ts):
type Stage = '规划中' | '研发中' | '公测灰度' | '商业化运营' | '已发布';
```

**stage → phase 映射 (L6,本 plan 实现):**
| stage | 对应 phase(s) | 含义 |
|-------|---------------|------|
| 规划中 | requirement | 需求阶段就绪率 |
| 研发中 | design + dev | 设计+开发阶段合计就绪率 |
| 公测灰度 | test | 测试阶段就绪率 |
| 商业化运营 | release | 发版阶段就绪率 |
| 已发布 | (全部) | 100% 视觉 |

**fallback 策略 (L5):**
- 优先读 `milestone.deliverableCodes`
- 若为 undefined 或空,尝试用 `milestone.deliverables` (旧字符串) 做模糊匹配:查 deliverables[].title.includes(legacyString) → 若匹配上,显示该 deliverable.status;若无匹配,显示"未关联"灰色徽章

   ```tsx
   {(() => {
     const codes = m.deliverableCodes && m.deliverableCodes.length > 0
       ? m.deliverableCodes
       : (m.deliverables || []);
     if (codes.length === 0) return null;

     const matched: FullLifecycleDeliverable[] = [];

           <span className="text-[10px] text-text-tertiary">{codes.length} 个未关联交付物</span>

             className={cn(

               d.status === 'ready' && 'bg-success/10 text-success',
               d.status === 'draft' && 'bg-bg-secondary text-text-tertiary',

     );
   })()}
   ```
