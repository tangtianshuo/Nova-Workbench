1. 新建 `src/lib/ai/associations.ts`(独立 wrapper,不强依赖 AppContext.arrangeOnCalendar — Phase 7 若已实施可后续 refactor 合并):
   ```typescript
   import { useTaskStore } from '@/src/stores/taskStore';
   import { useScheduleStore } from '@/src/stores/scheduleStore';

   export interface AssociationResult {
     success: boolean;
     eventId?: string;
     taskId?: string;
     error?: string;
     existingEventId?: string;
   }

   export function associateTaskWithEvent(
     taskId: string,
     date: string,
     time?: string,
   ): AssociationResult {
     const taskStore = useTaskStore.getState();
     const scheduleStore = useScheduleStore.getState();

     // Find the task
     const task = taskStore.categories
       .flatMap(c => c.tasks)
       .find(t => t.id === taskId);
     if (!task) return { success: false, error: `Task ${taskId} not found` };

     // Check existing association (idempotent guard)
     if (task.scheduledEventId) {
       return {
         success: false,
         error: 'Task already associated with an event',
         existingEventId: task.scheduledEventId,
         taskId,
       };
     }

     // Create event with back-reference
     const eventId = crypto.randomUUID();
     scheduleStore.addEvent({
       id: eventId,
       title: task.title,
       time: time ?? '',
       date,
       type: 'task',
       location: '',
       taskId,          // Phase 6 弱关联字段
     });

     // Set forward reference on task
     taskStore.updateTask(taskId, { scheduledEventId: eventId });

     return { success: true, eventId, taskId };
   }

   export function dissociateTaskEvent(taskId: string): AssociationResult {
     const taskStore = useTaskStore.getState();
     const task = taskStore.categories.flatMap(c => c.tasks).find(t => t.id === taskId);
     if (!task?.scheduledEventId) return { success: false, error: 'No association' };
     const eventId = task.scheduledEventId;
     taskStore.updateTask(taskId, { scheduledEventId: undefined });
     return { success: true, eventId, taskId };
   }
   ```

   ```typescript
   import { z } from 'zod';
   import { tool } from '../tool';
   import { useScheduleStore } from '@/src/stores/scheduleStore';
   import { useTaskStore } from '@/src/stores/taskStore';
   import { useProductStore } from '@/src/stores/productStore';
   import { useRndStore } from '@/src/stores/rndStore';
   import { associateTaskWithEvent } from '../associations';

   export const createEvent = tool({
     name: 'createEvent',
     description: 'Create a new calendar event. Example: 创建一个下周三的会议',
     schema: z.object({
       title: z.string(),
       date: z.string().describe('YYYY-MM-DD'),
       time: z.string().describe('e.g. "10:00 - 11:30"').optional(),
       type: z.enum(['meeting', 'review', 'sync', 'task', 'deadline', 'other']).default('other'),
       location: z.string().optional(),
     }),
     execute: async (args) => {
       const id = crypto.randomUUID();
       useScheduleStore.getState().addEvent({ id, ...args });
       return { success: true, eventId: id };
     },
     metadata: { destructive: false },
   });

   export const updateEvent = tool({
     name: 'updateEvent',
     description: 'Update fields of an existing calendar event',
     schema: z.object({
       eventId: z.string(),
       fields: z.object({
         title: z.string().optional(),
         date: z.string().optional(),
         time: z.string().optional(),
         location: z.string().optional(),
       }),
     }),
     execute: async ({ eventId, fields }) => {
       useScheduleStore.getState().updateEvent(eventId, fields);
       return { success: true, eventId };
     },
     metadata: { destructive: false },
   });

   export const deleteEvent = tool({
     name: 'deleteEvent',
     description: 'Permanently delete a calendar event. DESTRUCTIVE — requires confirmation. Also clears task back-references.',
     schema: z.object({ eventId: z.string() }),
     execute: async ({ eventId }) => {
       const store = useScheduleStore.getState();
       const event = store.events.find(e => e.id === eventId);
       if (event?.taskId) {
         // Clean back-reference on task
         useTaskStore.getState().updateTask(event.taskId, { scheduledEventId: undefined });
       }
       store.deleteEvent(eventId);
       return { success: true, eventId, deleted: true };
     },
     metadata: { destructive: true },
   });

   export const listEvents = tool({
     name: 'listEvents',
     description: 'List calendar events, optionally filtered by date range. Example: 这周有什么会',
     schema: z.object({
       startDate: z.string().describe('YYYY-MM-DD inclusive').optional(),
       endDate: z.string().describe('YYYY-MM-DD inclusive').optional(),
     }),
     execute: async ({ startDate, endDate }) => {
       let events = useScheduleStore.getState().events;
       if (startDate) events = events.filter(e => e.date >= startDate!);
       if (endDate) events = events.filter(e => e.date <= endDate!);
       events = [...events].sort((a, b) => a.date.localeCompare(b.date));
       return { success: true, count: events.length, events };
     },
     metadata: { destructive: false },
   });

   export const associateTaskToEvent = tool({
     name: 'associateTaskWithEvent',
     description: 'Schedule a task onto the calendar — creates a type="task" event and links task↔event bidirectionally. Example: 把这个任务安排到下周三',
     schema: z.object({
       taskId: z.string(),
       date: z.string().describe('YYYY-MM-DD — must resolve relative dates before calling'),
       time: z.string().optional(),
     }),
     execute: async ({ taskId, date, time }) => associateTaskWithEvent(taskId, date, time),
     metadata: { destructive: false },
   });

   export const getTaskDependencies = tool({
     name: 'getTaskDependencies',
     description: 'Get context for smart deadline suggestions — returns the task, its associated product, and its calendar event (if any).',
     schema: z.object({ taskId: z.string() }),
     execute: async ({ taskId }) => {
       const task = useTaskStore.getState().categories.flatMap(c => c.tasks).find(t => t.id === taskId);
       if (!task) return { success: false, error: 'Task not found' };
       const product = task.projectId
         ? useProductStore.getState().products.find(p => p.id === task.projectId) ?? null
         : null;
       const event = task.scheduledEventId
         ? useScheduleStore.getState().events.find(e => e.id === task.scheduledEventId) ?? null
         : null;
       return { success: true, task, product, event };
     },
     metadata: { destructive: false },
   });

   export const getProductFeatureBreakdown = tool({
     name: 'getProductFeatureBreakdown',
     description: 'Get milestones and R&D deliverables for a product — used to suggest feature breakdown. Example: 帮我拆解这个产品的功能矩阵',
     schema: z.object({ productId: z.string() }),
     execute: async ({ productId }) => {
       const product = useProductStore.getState().products.find(p => p.id === productId);
       if (!product) return { success: false, error: 'Product not found' };
       const deliverables = useRndStore.getState().deliverables[productId] ?? [];
       return {
         success: true,
         product: { id: product.id, name: product.name, tagline: product.tagline, stage: product.stage },
         milestones: product.milestones ?? [],
         deliverables: deliverables.map(d => ({
           code: d.code, name: d.name, phase: d.phase, status: d.status,
         })),
       };
     },
     metadata: { destructive: false },
   });

   export const scheduleTools = [
     createEvent, updateEvent, deleteEvent, listEvents,
     associateTaskToEvent, getTaskDependencies, getProductFeatureBreakdown,
   ];
   ```
