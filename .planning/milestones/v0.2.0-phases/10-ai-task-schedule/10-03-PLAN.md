   export function buildDateContext(now: Date = new Date()): string {
     const today = formatDate(now);
     const tomorrow = formatDate(addDays(now, 1));
     const nextMonday = formatDate(nextWeekday(now, 1));  // 0=Sun, 1=Mon
     const nextSunday = formatDate(nextWeekday(now, 0, true)); // 真正的下个周日
     const endOfMonth = formatDate(endOfMonth(now));

     const weekdayName = now.toLocaleDateString('en-US', { weekday: 'long' });

     return `## Current Date Context
   Today: ${today} (${weekdayName})
   Tomorrow: ${tomorrow}
   Next Monday: ${nextMonday}
   Next Sunday: ${nextSunday}
   End of month: ${endOfMonth}

   ## Relative Date Resolution Rules
   - "明天" → ${tomorrow}
   - "后天" → today + 2 days
   - "下周X" → next week's weekday X (Monday=周一 ... Sunday=周日)
   - "本周末" → this Saturday or Sunday
   - "N 天后" → today + N days
   - "月底" → ${endOfMonth}
   IMPORTANT: Resolve relative dates YOURSELF using the rules above before calling tools that require YYYY-MM-DD. Never pass raw "下周三" to a tool.`;
   }

   function formatDate(d: Date): string {
     return d.toISOString().slice(0, 10);
   }
   function addDays(d: Date, n: number): Date {
     const r = new Date(d); r.setDate(r.getDate() + n); return r;
   }
   function nextWeekday(d: Date, target: number, nextWeekOnly = false): Date {
     const r = new Date(d);
     const cur = r.getDay();
     let diff = (target - cur + 7) % 7;
     if (diff === 0 && nextWeekOnly) diff = 7;  // 同一天且要求"下周"则 +7
     if (nextWeekOnly && diff === 0) diff = 7;
     if (!nextWeekOnly && diff === 0) diff = 7;  // "下周三" 若今天就是周三,默认指下周
     r.setDate(r.getDate() + diff);
     return r;
   }
   function endOfMonth(d: Date): Date {
     return new Date(d.getFullYear(), d.getMonth() + 1, 0);
   }
   ```
   (注:`nextWeekday` 逻辑在 `nextWeekOnly=true` 时统一 +7:对于 "下周X" 场景,即使是同一天也跳到下周。)

2. 新建 `src/lib/ai/chatSession.ts`:

   export interface ChatMessage {
     role: 'user' | 'assistant' | 'tool';
     content: string;
     toolCallId?: string;
     toolName?: string;
     timestamp: number;
   }

   export class ChatSession {
     private messages: ChatMessage[] = [];
     readonly sessionId: string;

     constructor(sessionId?: string) {
       this.sessionId = sessionId ?? crypto.randomUUID();
     }

     addMessage(msg: Omit<ChatMessage, 'timestamp'>): void {
       this.messages.push({ ...msg, timestamp: Date.now() });
     }

     /** 最近 N 轮 (1 轮 = user + assistant pair,tool messages 折叠进 assistant) */
     getMessagesForLLM(maxTurns = 8): Array<{ role: string; content: string }> {
       const pairs: ChatMessage[][] = [];
       let cur: ChatMessage[] = [];
       for (const m of this.messages) {
         cur.push(m);
         if (m.role === 'assistant' && !m.toolCallId) {
           pairs.push(cur); cur = [];
         }
       }
       if (cur.length) pairs.push(cur);
       const recent = pairs.slice(-maxTurns);
       return recent.flat().map(m => ({ role: m.role, content: m.content }));
     }

     getAllMessages(): ChatMessage[] {
       return [...this.messages];
     }

     clear(): void {
       this.messages = [];
     }

     /** Rough token estimate (content length / 4) */
     estimateTokens(): number {
       return Math.ceil(this.messages.reduce((s, m) => s + m.content.length, 0) / 4);
     }
   }
   ```
