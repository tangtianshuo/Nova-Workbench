带 belt-and-braces 守卫的用法封装（同文件导出，供两区使用）：
```tsx
export function AiContextMenu({ children, items }: {
  children: ReactElement;
  items: Array<{ icon: ReactNode; label: string; onSelect: () => void }>;
}) {
  const [open, setOpen] = useState(false);
  return (
    <ContextMenu open={open} onOpenChange={setOpen}>
      <ContextMenuTrigger asChild onContextMenu={(e) => {
        if ((e.target as HTMLElement).closest('[contenteditable="true"]')) setOpen(false);
      }}>{children}</ContextMenuTrigger>
      <ContextMenuContent>
        <ContextMenuLabel>AI 动作</ContextMenuLabel>
        {items.map((it) => <ContextMenuItem key={it.label} onSelect={it.onSelect}>{it.icon}{it.label}</ContextMenuItem>)}
      </ContextMenuContent>
    </ContextMenu>
  );
}
```
受控 open=false 拦截 contenteditable 目标的开菜单（主守卫是结构性的：编辑器面板从不包裹 — 锁定）。

export function fireAiAction(instruction: string): void {
  const selection = window.getSelection()?.toString().trim() ?? '';
  const snapshot = selection
    ? `引用选区：「${selection.length > 200 ? selection.slice(0, 200) + '…' : selection}」\n\n`
    : '';
  const s = useUIStore.getState();
  s.setPendingChatPrefill(snapshot + instruction);  // 覆盖式（每次点击都是新意图）
  s.setChatPanelOpen(true);                          // Drawer 叠在当前视图上，不切 tab
}
