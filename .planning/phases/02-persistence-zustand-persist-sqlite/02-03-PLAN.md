Pattern for wrapping (RESEARCH.md "Pattern 3"):
```ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import { sqliteStorage } from './storage/sqliteStorage';

export const useXStore = create<XState>()(
  persist(
    (set, get) => ({
      // existing state + actions
      _hasHydrated: false,
      _setHydrated: () => set({ _hasHydrated: true }),
    }),
    {
      name: 'nova-x',
      version: 1,
      storage: sqliteStorage,
      partialize: (s) => ({ /* only serializable fields */ }),
      migrate: (persisted, _version) => persisted as Partial<XState>,
      onRehydrateStorage: () => (state) => {
        state?._setHydrated();
      },
    },
  ),
);
```

Important: TypeScript needs the state interface to include `_hasHydrated: boolean` and `_setHydrated: () => void` BEFORE we add them to the create() body. Update interface first.
