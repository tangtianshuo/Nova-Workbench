     **更简单实现**: 直接把内部组件放进单独文件 `MarkdownEditorInner.tsx` (同目录),MarkdownEditor.tsx 用 `lazy(() => import('./MarkdownEditorInner'))`. 这样 MarkdownEditor.tsx 本身不 import @mdxeditor/editor,保持 barrel re-export 时不拖入主 bundle.

     ```typescript
     interface MarkdownEditorProps {
       value: string;
       onChange: (markdown: string) => void;
       readOnly?: boolean;
       placeholder?: string;
       className?: string;
       minHeight?: string; // 默认 '320px'
     }
     export const MarkdownEditor = forwardRef<HTMLDivElement, MarkdownEditorProps>(function MarkdownEditor(
       { value, onChange, readOnly = false, placeholder, className, minHeight = '320px' },
       _ref
     ) {
       return (
         <Suspense fallback={<Skeleton className={cn('w-full', className)} style={{ minHeight }} />}>
           <MarkdownEditorInner
             value={value}
             onChange={onChange}
             readOnly={readOnly}
             placeholder={placeholder}
             className={className}
             minHeight={minHeight}
           />
         </Suspense>
       );
     });
     MarkdownEditor.displayName = 'MarkdownEditor';
     ```
