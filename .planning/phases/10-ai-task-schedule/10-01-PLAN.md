   ```typescript
   import { z } from 'zod';
   import { tool } from '../tool';
   import { useTaskStore } from '@/src/stores/taskStore';

   export const updateTask = tool({
     name: 'updateTask',
     description: 'Update one or more fields of an existing task. Use this for editing title, priority, status, description, or deadline. Example: 把任务优先级改为低',
     schema: z.object({
       taskId: z.string().describe('The unique task ID (UUID format)'),
       fields: z.object({
         title: z.string().optional(),
         priority: z.enum(['high', 'medium', 'low']).optional(),
         status: z.enum(['未开始', '进行中', '已完成']).optional(),
         description: z.string().optional(),
         deadline: z.string().describe('YYYY-MM-DD format').optional(),
       }),
     }),
     execute: async ({ taskId, fields }) => {
       useTaskStore.getState().updateTask(taskId, fields);
       return { success: true, taskId };
     },
     metadata: { destructive: false },
   });

   export const deleteTask = tool({
     name: 'deleteTask',
     description: 'Permanently delete a task. DESTRUCTIVE — requires user confirmation. Example: 删除那个重复任务',
     schema: z.object({ taskId: z.string() }),
     execute: async ({ taskId }) => {
       useTaskStore.getState().deleteTask(taskId);
       return { success: true, taskId, deleted: true };
     },
     metadata: { destructive: true },  // Phase 9 tool loop 检查此 flag 插入确认 (D-07)
   });

   export const moveTask = tool({
     name: 'moveTask',
     description: 'Move a task to a different category column (kanban lane). Example: 把这个任务挪到开发实现',
     schema: z.object({
       taskId: z.string(),
       toCategoryId: z.string().describe('Target category ID'),
     }),
     execute: async ({ taskId, toCategoryId }) => {
       const cats = useTaskStore.getState().categories;
       const fromCat = cats.find(c => c.tasks.some(t => t.id === taskId));
       if (!fromCat) return { success: false, error: 'Task not found' };
       useTaskStore.getState().moveTask(taskId, fromCat.id, toCategoryId);
       return { success: true, taskId, fromCategoryId: fromCat.id, toCategoryId };
     },
     metadata: { destructive: false },
   });

   export const rescheduleTask = tool({
     name: 'rescheduleTask',
     description: 'Change the deadline of a task. Accepts YYYY-MM-DD only — caller must resolve relative dates (下周三/明天) before calling. Example: 把这个任务的截止日期改到 2026-08-15',
     schema: z.object({
       taskId: z.string(),
       newDate: z.string().describe('YYYY-MM-DD format'),
     }),
     execute: async ({ taskId, newDate }) => {
       useTaskStore.getState().updateTask(taskId, { deadline: newDate });
       return { success: true, taskId, newDeadline: newDate };
     },
     metadata: { destructive: false },
   });

   export const setTaskPriority = tool({
     name: 'setTaskPriority',
     description: 'Set the priority of a task. Example: 把任务优先级改为高',
     schema: z.object({
       taskId: z.string(),
       priority: z.enum(['high', 'medium', 'low']),
     }),
     execute: async ({ taskId, priority }) => {
       useTaskStore.getState().updateTask(taskId, { priority });
       return { success: true, taskId, priority };
     },
     metadata: { destructive: false },
   });

   export const taskTools = [updateTask, deleteTask, moveTask, rescheduleTask, setTaskPriority];
   ```

   ```typescript
   import { z } from 'zod';
   import { tool } from '../tool';
   import { useTaskStore } from '@/src/stores/taskStore';

   export const bulkCompleteTasks = tool({
     name: 'bulkCompleteTasks',
     description: 'Mark multiple tasks as completed at once. Example: 把这几个任务都标记为已完成',
     schema: z.object({ taskIds: z.array(z.string()).min(1).max(50) }),
     execute: async ({ taskIds }) => {
       const failed: string[] = [];
       let completed = 0;
       const store = useTaskStore.getState();
       const existingIds = new Set(store.categories.flatMap(c => c.tasks.map(t => t.id)));
       for (const id of taskIds) {
         if (existingIds.has(id)) { store.completeTask(id); completed++; }
         else failed.push(id);
       }
       return { success: true, completed, failed };
     },
     metadata: { destructive: false },
   });

   export const bulkDeleteTasks = tool({
     name: 'bulkDeleteTasks',
     description: 'Permanently delete multiple tasks. DESTRUCTIVE — requires user confirmation. Example: 删除这些重复任务',
     schema: z.object({ taskIds: z.array(z.string()).min(1).max(50) }),
     execute: async ({ taskIds }) => {
       const failed: string[] = [];
       let deleted = 0;
       const store = useTaskStore.getState();
       const existingIds = new Set(store.categories.flatMap(c => c.tasks.map(t => t.id)));
       for (const id of taskIds) {
         if (existingIds.has(id)) { store.deleteTask(id); deleted++; }
         else failed.push(id);
       }
       return { success: true, deleted, failed };
     },
     metadata: { destructive: true },
   });

   export const bulkUpdatePriority = tool({
     name: 'bulkUpdatePriority',
     description: 'Update the priority of multiple tasks at once. Example: 把所有高优先级任务改为中优先级',
     schema: z.object({
       taskIds: z.array(z.string()).min(1).max(50),
       priority: z.enum(['high', 'medium', 'low']),
     }),
     execute: async ({ taskIds, priority }) => {
       const failed: string[] = [];
       let updated = 0;
       const store = useTaskStore.getState();
       const existingIds = new Set(store.categories.flatMap(c => c.tasks.map(t => t.id)));
       for (const id of taskIds) {
         if (existingIds.has(id)) { store.updateTask(id, { priority }); updated++; }
         else failed.push(id);
       }
       return { success: true, updated, failed };
     },
     metadata: { destructive: false },
   });

   export const bulkTools = [bulkCompleteTasks, bulkDeleteTasks, bulkUpdatePriority];
   ```
