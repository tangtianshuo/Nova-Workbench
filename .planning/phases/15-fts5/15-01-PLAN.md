```typescript
export function toFtsTokens(text: string): string[] {
  const normalized = text.toLocaleLowerCase().normalize('NFKC');
  const words = normalized.match(/[a-z0-9]+/g) ?? [];
  const cjk = normalized.match(/[㐀-鿿]/g) ?? [];
  return [...new Set([...words, ...cjk])];
}
export function toFtsIndexedText(text: string): string {
  return toFtsTokens(text).join(' ');
}
export function toFtsMatchString(query: string): string {
  return toFtsTokens(query).map((t) => `"${t.replace(/"/g, '""')}"`).join(' ');
}
```

export const MEMORY_CANDIDATE_TTL_MS = 7 * 24 * 60 * 60 * 1000; // 7 天（锁定决策）
export const MEMORY_CANDIDATE_CAP = 20;                          // ~20 上限（锁定决策）
export type MemoryCandidateOrigin = 'model_inferred' | 'user_directed';

propose(input: { content: string; origin: MemoryCandidateOrigin; scope: 'global'|'product'; productId?: string; sessionId?: string }): Promise<ProposeResult>
// ProposeResult = { ok: true, candidateToken: string, deduplicated: boolean, reason?: 'duplicate_pending'|'previously_rejected', evictedOldest: boolean }
// 实现: content_hash = await computeParamsHash({ content, scope, productId }) — 照 paramsHash 复用（锁定）
// SQLite 实现 propose 流程（每步单语句，无事务）:
//   1) SELECT status FROM memory_candidates WHERE content_hash=$1 — 命中 pending/confirmed(未过期) → deduplicated
//      命中 rejected → deduplicated + reason:'previously_rejected'（Open Question 3: origin='user_directed' 放行复活 — INSERT 前置 UPDATE 该行回 pending）
//   2) SELECT COUNT(*) WHERE status='pending' AND expires_at>$now — >= CAP 则执行 RESEARCH「cap 满时让位」SQL:
//      UPDATE memory_candidates SET expires_at=$now WHERE candidate_token=(SELECT candidate_token FROM memory_candidates WHERE status='pending' AND expires_at>$now ORDER BY created_at ASC LIMIT 1)
//   3) INSERT pending 行 (candidate_token=crypto.randomUUID(), expires_at=now+7d)
//   4) stats 计数器 dedupHits/evictions 内存累加（单例字段）

confirm(token) / reject(token) / listPending(): 照抄 confirmationStore.confirm/reject/listActive 的条件 UPDATE + 派生过期
consumeIntoMemories(token): 条件 UPDATE status='confirmed'→'consumed'（rowsAffected!==1 抛 already_settled）后 INSERT INTO memories
insertMemory(input: { memoryId?: string; content: string; origin; scope; productId?; sourceSessionId?; sourceCandidateToken?; supersedesRowid?: number }): Promise<MemoryRecord>
// version = 同 memory_id 当前最大 version+1（SELECT MAX(version)）；若 supersedesRowid 提供 → 单语句 UPDATE memories SET superseded_at=$now WHERE memory_rowid=$r AND superseded_at IS NULL
listActiveMemories(productId?): WHERE superseded_at IS NULL AND deleted_at IS NULL AND (scope='global' OR product_id=$pid) ORDER BY confirmed_at DESC
listAllMemories(): 不过滤（UI 记忆列表含「已被取代」标记）
deleteMemory(memoryRowid): UPDATE memories SET deleted_at=$now WHERE memory_rowid=$r AND deleted_at IS NULL
deleteByProduct(productId): UPDATE memories SET deleted_at=$now WHERE product_id=$1 AND deleted_at IS NULL
listRejected(limit=10): status='rejected' ORDER BY rejected_at DESC（MEM-02 拒绝列表，供组装器注入系统提示）
stats(): { pendingCount, dedupHits, evictions }
