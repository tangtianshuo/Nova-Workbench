```typescript
import { z } from 'zod';
import { registerTool } from '../registry';
import { getMemoryStore } from '../memoryStore';

const proposeMemorySchema = z.object({
  content: z.string().trim().min(1).max(500),
  scope: z.enum(['global', 'product']).optional(),
  userDirected: z.boolean().optional(), // 用户显式说"记住"时模型置 true → origin='user_directed'
}).strict();

registerTool({
  name: 'proposeMemory',
  description: 'Propose a long-term memory candidate from this conversation. Never fabricates user confirmation; the user must approve the candidate in the UI. Use userDirected=true ONLY when the user explicitly said to remember something.',
  schema: proposeMemorySchema,
  execute: async (args) => {
    const store = getMemoryStore();
    const productId = args.scope === 'product' ? useUIStore.getState().selectedProductId : undefined;
    const result = await store.propose({
      content: args.content,
      origin: args.userDirected ? 'user_directed' : 'model_inferred',
      scope: args.scope ?? 'global',
      productId,
      sessionId: getCurrentSessionId(), // 从 uiStore 或模块级 current session 变量取；无则 undefined
    });
    return {
      ok: true,
      candidateQueued: !result.deduplicated,
      ...result,
      note: 'The user will see this candidate in the chat panel; do not re-propose the same content.',
    };
  },
});
```

红线（RESEARCH Pattern 4，Anti-Patterns 第 1 条）：本 tool **绝不** import 或抛 `ConfirmationRequiredError` — 那条路径 endTurn('awaiting_confirmation') 会终止对话。也不往 `agent_confirmation_candidates` 表插任何行（Pitfall 2）。sessionId 获取若 chatPanel 链路无现成通道，就先 `sessionId: undefined`（候选恢复走 memoryStore.listPending 全量，不依赖 session 过滤 — 最省事且满足 UI spec Surface 1 restore 行为）。

```typescript
export interface SegmentAudit { name: 'core'|'pending'|'memories'|'fts_topk'|'recent_dialog'; items: number; tokens: number; truncated: boolean; reservedTokens?: number; query?: string; skipped?: boolean; error?: string }
export interface AssembledContext { coreContext: string; audit: { segments: SegmentAudit[] } }

export async function assembleInjectedContext(opts: {
  buildCore: () => string;                      // 注入 buildCoreContext（纯函数可测）
  searchKnowledge: (query: string, limit: number) => Promise<KnowledgeSearchHit[]>; // 注入 knowledgeRepo.search（15-03 前传 mock；hit = { title, version, updatedAt, summary }）
  productId?: string | null;
  userMessage: string;
}): Promise<AssembledContext>
```

实现（锁定决策 RESEARCH Pattern 5）：
- 配额（固定比例，基数 session 上下文 8000）：core 30%（=2400）→ 但**总注入硬上限 2000 tokens** 兜底，各段实际配额 = min(ratio×8000, 按段分摊 2000)。具体：段配额 = [600, 200, 500, 400, 300]（core/pending/memories/fts/recent 预留），合计 2000 = 25% of 8000，同时保持 30/10/25/20/15 比例（600:200:500:400:300）。段内超配额 → 从最旧条目开始丢弃，标记 truncated。
- 段组装顺序（注入文本，最终拼进 coreContext 字符串返回）：
  1. core：`opts.buildCore()` 原样（锁定：保留为业务事实段输入）
  2. pending：`listPending()` 条目 + `listRejected(5)` 条目，rejected 渲染为 `不要再提出以下记忆（用户已拒绝）: ...`
  3. memories：`listActiveMemories(opts.productId)` 每条一行
  4. fts_topk：`toFtsTokens(userMessage)` 空则 skipped；否则 `opts.searchKnowledge(toFtsMatchString(userMessage), 5)`（top-k=5 锁定），每条带 `来源: {title} v{version} {updatedAt}`；异常 catch 为 error:'unavailable'
  5. recent_dialog：不注入内容，仅 `reservedTokens: 1200`（=15% of 8000）审计记账
- 计量全用 `estimateTokens`。
- `context_injected` 事件追加**不在本模块**（纯函数可测）；事件由 Task 3 在 toolLoop 侧落，payload = `audit`。

若 `src/ai/events/eventStore.ts` 的 eventType 类型联合不含 `'context_injected'`，本任务顺手把该字面量加进联合类型（一行）。
