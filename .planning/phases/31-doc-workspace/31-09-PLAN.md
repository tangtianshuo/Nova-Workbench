2) 在 MarkdownEditorInner.tsx 新增 `codeBlockView` 插件并 `.use()` 进链(与 livePreviewPlugin 同层)。用 `$view(codeBlockSchema, (ctx) => view)` 覆盖 code_block 节点的 ProseMirror NodeView,**纯 DOM 构建(document.createElement),不走 ReactDOM/React 组件** —— NodeView 生命周期由 ProseMirror 管理,React 封装边界(@milkdown/react)只管编辑器整体,别把 React root 塞进 NodeView:

```ts
const LANGS = ['plain', 'typescript', 'javascript', 'json', 'bash', 'rust', 'python', 'markdown', 'sql', 'yaml', 'xml', 'css'];
// 语言包 side-effect import(prism-core 之外按需):
import 'prismjs/components/prism-typescript';
import 'prismjs/components/prism-json';
import 'prismjs/components/prism-bash';
import 'prismjs/components/prism-rust';
import 'prismjs/components/prism-python';
import 'prismjs/components/prism-markdown';
import 'prismjs/components/prism-sql';
import 'prismjs/components/prism-yaml';
// (javascript/css/markup 是 prismjs 核心自带依赖,typescript 依赖 javascript)

const codeBlockView = $view(codeBlockSchema, () => (node, view, getPos) => {
  const dom = document.createElement('div');
  dom.className = 'milkdown-code-block';
  const bar = document.createElement('div');
  bar.className = 'milkdown-code-bar';
  const barText = document.createElement('span');  // 语言名展示
  const select = document.createElement('select'); // 语言下拉(原生 select,Nova tokens 样式)
  select.className = 'milkdown-code-lang';
  for (const l of LANGS) { const o = document.createElement('option'); o.value = l; o.textContent = l; select.appendChild(o); }
  select.value = String(node.attrs.language || 'plain');
  select.addEventListener('mousedown', (e) => e.stopPropagation()); // 防编辑器失焦/拖选
  select.addEventListener('change', () => {
    const pos = typeof getPos === 'function' ? getPos() : undefined;
    if (typeof pos !== 'number') return;
    const language = select.value === 'plain' ? '' : select.value;
    view.dispatch(view.state.tr.setNodeMarkup(pos, undefined, { ...node.attrs, language }));
  });
  bar.append(barText, select);
  const pre = document.createElement('pre');
  const code = document.createElement('code');
  code.className = 'milkdown-code-content';
  pre.appendChild(code);
  dom.append(bar, pre);
  const highlight = (n: typeof node) => {
    code.textContent = n.textContent;
    const lang = String(n.attrs.language || '');
    code.className = 'milkdown-code-content' + (lang ? ` language-${lang}` : '');
    if (lang && Prism.languages[lang]) Prism.highlightElement(code);
  };
  highlight(node);
  return {
    dom,
    update: (n) => { if (n.type.name !== 'code_block') return false; node = n; highlight(n); return true; },
    // code_block 无嵌套 content 变更时 update 足够;selectMutation 由 ProseMirror 默认处理
  } satisfies NodeView;
});
```

(以上为骨架,executor 按实际类型签名适配;`setNodeMarkup` 是 prosemirror 标准路径,不绕 `updateCodeBlockLanguageCommand` —— 该命令需 ctx 闭包,NodeView 内 dispatch tr 更直接。readonly 态:select 加 `disabled = !view.editable`。)
